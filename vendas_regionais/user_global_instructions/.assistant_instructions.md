```python

    """
    SEMPRE executar antes de criar/modificar arquivos.

    raiz_esperada = f"/Workspace/Users/{username}/data-in-code/{projeto}/"
    
    if not path.startswith(raiz_esperada):
        raise ValueError(
            f"❌ PATH INVÁLIDO!\n"
            f"Expected: {raiz_esperada}\n"
            f"Got: {path}\n"
            f"TODOS os paths devem estar dentro de data-in-code/{projeto}/"
        )
    
    return True
```
